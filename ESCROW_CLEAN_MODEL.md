## The Escrow Service Model

### What We Provide:

**1. Transparent Blockchain Escrow**
- Funds held in smart contract (auditable 24/7)
- Parties can verify reserves on blockchain
- Automated release per agreement terms
- Dispute resolution built-in

**2. Multi-Party Settlement**
- Support complex deals (buyer, seller, intermediaries)
- Milestone-based releases (e.g., 50% on shipment, 50% on delivery)
- Conditional logic (IF shipment confirmed THEN release)

**3. Full Compliance**
- Segregated custody (client funds separate)
- Regular audits by external firm
- Real-time reporting to regulators
- KYC/AML on all parties

**4. Lower Cost**
- 0.5% service fee (vs 2-5% traditional escrow)
- No hidden fees
- Transparent pricing

### What We DON'T Do (To Stay Clean):

❌ **NO investment of escrow funds** - Funds stay liquid
❌ **NO commingling** - Client funds never mixed with operating capital
❌ **NO fractional reserve** - Always 100% backed
❌ **NO yield games** - Any interest earned goes to client
❌ **NO proprietary trading** - We don't trade with client money

## Technical Implementation

### Smart Contract Architecture:

**contracts/EscrowVault.sol**
```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/access/AccessControl.sol";
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";

/**
 * Grain & Vault Escrow Contract
 *
 * Key Features:
 * - Segregated custody (each escrow is isolated)
 * - Multi-party support (buyer, seller, arbitrator)
 * - Milestone-based releases
 * - Dispute resolution
 * - Transparent on-chain tracking
 */
contract EscrowVault is AccessControl, ReentrancyGuard {
    bytes32 public constant ADMIN_ROLE = keccak256("ADMIN_ROLE");
    bytes32 public constant ORACLE_ROLE = keccak256("ORACLE_ROLE");

    IERC20 public usdc;

    enum EscrowStatus { Active, Completed, Disputed, Cancelled, Expired }

    struct Milestone {
        string description;
        uint256 percentage;  // Percentage of total to release
        bool completed;
        uint256 completedAt;
    }

    struct Escrow {
        // Parties
        address depositor;
        address beneficiary;
        address arbitrator;

        // Amounts
        uint256 totalAmount;
        uint256 releasedAmount;

        // Milestones
        Milestone[] milestones;

        // Status
        EscrowStatus status;

        // Timestamps
        uint256 createdAt;
        uint256 expiresAt;
        uint256 completedAt;

        // Metadata
        string dealId;  // External reference (e.g., "DR-$3B-2025")
    }

    mapping(uint256 => Escrow) public escrows;
    uint256 public escrowCount;

    // Events
    event EscrowCreated(uint256 indexed escrowId, address depositor, address beneficiary, uint256 amount);
    event MilestoneCompleted(uint256 indexed escrowId, uint256 milestoneIndex, uint256 amountReleased);
    event EscrowCompleted(uint256 indexed escrowId, uint256 totalReleased);
    event EscrowDisputed(uint256 indexed escrowId, address disputedBy);
    event DisputeResolved(uint256 indexed escrowId, address resolvedBy);

    constructor(address _usdc) {
        usdc = IERC20(_usdc);
        _setupRole(DEFAULT_ADMIN_ROLE, msg.sender);
        _setupRole(ADMIN_ROLE, msg.sender);
    }

    /**
     * Create new escrow
     */
    function createEscrow(
        address beneficiary,
        address arbitrator,
        uint256 amount,
        uint256 durationDays,
        string memory dealId,
        Milestone[] memory milestones
    ) external nonReentrant returns (uint256) {
        require(amount > 0, "Amount must be positive");
        require(beneficiary != address(0), "Invalid beneficiary");
        require(milestones.length > 0, "Need at least one milestone");

        // Transfer USDC from depositor to this contract
        require(
            usdc.transferFrom(msg.sender, address(this), amount),
            "USDC transfer failed"
        );

        escrowCount++;
        Escrow storage escrow = escrows[escrowCount];

        escrow.depositor = msg.sender;
        escrow.beneficiary = beneficiary;
        escrow.arbitrator = arbitrator;
        escrow.totalAmount = amount;
        escrow.releasedAmount = 0;
        escrow.status = EscrowStatus.Active;
        escrow.createdAt = block.timestamp;
        escrow.expiresAt = block.timestamp + (durationDays * 1 days);
        escrow.dealId = dealId;

        // Add milestones
        for (uint i = 0; i < milestones.length; i++) {
            escrow.milestones.push(milestones[i]);
        }

        emit EscrowCreated(escrowCount, msg.sender, beneficiary, amount);
        return escrowCount;
    }

    /**
     * Complete milestone (release funds)
     * Can be called by oracle or admin after verification
     */
    function completeMilestone(uint256 escrowId, uint256 milestoneIndex)
        external
        onlyRole(ORACLE_ROLE)
        nonReentrant
    {
        Escrow storage escrow = escrows[escrowId];
        require(escrow.status == EscrowStatus.Active, "Escrow not active");
        require(milestoneIndex < escrow.milestones.length, "Invalid milestone");

        Milestone storage milestone = escrow.milestones[milestoneIndex];
        require(!milestone.completed, "Milestone already completed");

        // Calculate release amount
        uint256 releaseAmount = (escrow.totalAmount * milestone.percentage) / 100;
        require(escrow.releasedAmount + releaseAmount <= escrow.totalAmount, "Exceeds total");

        // Mark milestone complete
        milestone.completed = true;
        milestone.completedAt = block.timestamp;

        // Release funds to beneficiary
        require(
            usdc.transfer(escrow.beneficiary, releaseAmount),
            "USDC transfer failed"
        );

        escrow.releasedAmount += releaseAmount;

        emit MilestoneCompleted(escrowId, milestoneIndex, releaseAmount);

        // Check if all milestones complete
        bool allComplete = true;
        for (uint i = 0; i < escrow.milestones.length; i++) {
            if (!escrow.milestones[i].completed) {
                allComplete = false;
                break;
            }
        }

        if (allComplete) {
            escrow.status = EscrowStatus.Completed;
            escrow.completedAt = block.timestamp;
            emit EscrowCompleted(escrowId, escrow.releasedAmount);
        }
    }

    /**
     * Raise dispute
     * Freezes all releases until resolved
     */
    function disputeEscrow(uint256 escrowId) external {
        Escrow storage escrow = escrows[escrowId];
        require(escrow.status == EscrowStatus.Active, "Escrow not active");
        require(
            msg.sender == escrow.depositor ||
            msg.sender == escrow.beneficiary,
            "Not authorized"
        );

        escrow.status = EscrowStatus.Disputed;
        emit EscrowDisputed(escrowId, msg.sender);
    }

    /**
     * Resolve dispute
     * Only arbitrator can call this
     */
    function resolveDispute(
        uint256 escrowId,
        uint256 amountToDepositor,
        uint256 amountToBeneficiary
    ) external nonReentrant {
        Escrow storage escrow = escrows[escrowId];
        require(escrow.status == EscrowStatus.Disputed, "Not disputed");
        require(msg.sender == escrow.arbitrator, "Not arbitrator");
        require(
            amountToDepositor + amountToBeneficiary <= escrow.totalAmount - escrow.releasedAmount,
            "Exceeds remaining amount"
        );

        // Send funds per arbitrator decision
        if (amountToDepositor > 0) {
            require(
                usdc.transfer(escrow.depositor, amountToDepositor),
                "Transfer to depositor failed"
            );
        }

        if (amountToBeneficiary > 0) {
            require(
                usdc.transfer(escrow.beneficiary, amountToBeneficiary),
                "Transfer to beneficiary failed"
            );
        }

        escrow.releasedAmount += amountToDepositor + amountToBeneficiary;
        escrow.status = EscrowStatus.Completed;
        escrow.completedAt = block.timestamp;

        emit DisputeResolved(escrowId, msg.sender);
        emit EscrowCompleted(escrowId, escrow.releasedAmount);
    }

    /**
     * Get escrow details
     */
    function getEscrow(uint256 escrowId) external view returns (
        address depositor,
        address beneficiary,
        uint256 totalAmount,
        uint256 releasedAmount,
        EscrowStatus status,
        uint256 createdAt,
        uint256 expiresAt
    ) {
        Escrow storage escrow = escrows[escrowId];
        return (
            escrow.depositor,
            escrow.beneficiary,
            escrow.totalAmount,
            escrow.releasedAmount,
            escrow.status,
            escrow.createdAt,
            escrow.expiresAt
        );
    }

    /**
     * Get milestone status
     */
    function getMilestones(uint256 escrowId) external view returns (Milestone[] memory) {
        return escrows[escrowId].milestones;
    }
}
```
